def get_top_weekly(chat_id: int):
    """Топ за последние 7 дней в конкретном чате"""
    cur.execute("""
        SELECT u.first_name, u.username, 
               SUM(b.payout - b.bet_amount) as weekly_balance,
               CAST((SUM(b.payout - b.bet_amount) * 1.0 / -10.0) + 100 AS INTEGER) as coo_result
        FROM bets b
        JOIN users u ON b.user_id = u.user_id AND b.chat_id = u.chat_id
        WHERE b.created_at >= datetime('now', '-7 days') AND b.chat_id = ?
        GROUP BY b.user_id, b.chat_id
        ORDER BY coo_result ASC
    """, (chat_id,))
    return cur.fetchall()

def decode_slot(value: int) -> tuple[str, str, str]:
    """value (1..64) → три символа на барабанах (четверичное представление)."""
    n = value - 1
    reels = []
    for _ in range(3):
        reels.append(SLOT_SYMBOLS[n % 4])
        n //= 4
    return tuple(reversed(reels))  # type: ignore[return-value]

def slot_payout(value: int, bet: int) -> tuple[int, str]:
    a, b, c = decode_slot(value)
    combo = f"{a} | {b} | {c}"
    if a == b == c:
        if a == "7️⃣":
            return bet * 10, "🎉 ДЖЕКПОТ 7️⃣7️⃣7️⃣! (x10)" # Снизили с 15 до 10
        if a == "BAR":
            return bet * 4, "🔥 BAR BAR BAR (x4)" # Снизили с 5 до 4
        if a == "🍋":
            return bet * 2, "✨ 🍋🍋🍋 (x2)"
        return int(bet * 1.5), "👍 🍒🍒🍒 (x1.5)" # Снизили с 2 до 1.5
    
    # УДАЛИЛИ БЛОК С ДВУМЯ ОДИНАКОВЫМИ (теперь это проигрыш)
    return 0, f"💨 {combo} — мимо"

def resolve_game(emoji: str, value: int, bet: int) -> tuple[int, str]:

    if emoji == "🎰":

    if emoji == "🎲":
        if value == 6: # Выигрыш только если выпадет 6
            return bet * 2, f"🎲 Выпало {value} — победа (x2)"
        return 0, f"🎲 Выпало {value} — проигрыш" # 4 и 5 теперь проигрыш
    if emoji == "🎯":
        if value == 6:
            return bet * 3, "🎯 В яблочко! (x3)" # Уменьшили выплату с x5 до x3
        return 0, f"🎯 {value} — мимо"
    if emoji == "🎳":

            return bet * 3, "🎳 СТРАЙК! (x3)"

            return bet * 2, f"⚽ {value} — гол! (x2)"
        return 0, f"⚽ {value} — мимо"
    return bet, "Неизвестная игра — возврат"

# ---------- Middleware: простой rate-limit (1 запрос/сек на юзера) ----------
class ThrottlingMiddleware:
    def __init__(self, rate_limit: float = 1.0):
        self.rate_limit = rate_limit
        self.last: dict[int, float] = defaultdict(float)

    async def __call__(self, handler, event, data):
        if not isinstance(event, Message) or event.from_user is None:
            return await handler(event, data)
        uid = event.from_user.id
        now = monotonic()
        if now - self.last[uid] < self.rate_limit:
            return  # тихо дропаем слишком частые сообщения
        self.last[uid] = now
        return await handler(event, data)

dp.message.outer_middleware(ThrottlingMiddleware(1.0))

            await message.reply("Использование: /add_balance @ник 500 (или ответьте на сообщение игрока)")
            return
        target = get_user_id_by_arg(args[1])
        if not target:
            await message.reply("❌ Игрок с таким ником не найден. (Он должен сначала написать боту).")
            return
        try:
            amount = int(args[2])
        except ValueError:
            await message.reply("❌ Сумма должна быть числом.")
            return

    add_transaction(target, "admin_add", amount, f"admin={message.from_user.id}")
    await message.reply(
        f"✅ Начислено <b>{amount}</b> игроку <code>{target}</code>.\n"
        f"Его баланс: <b>{get_balance(target)}</b>",
        parse_mode="HTML",
    )
